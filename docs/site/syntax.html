ALX Language Syntax

Complete reference for ALX 0.4.0

Hello World

print("Hello, ALEXION!")

Comments

// This is a comment

Variables

name = "ALX"
version = 1
health = 100
alive = true
value = null

// Reassignment
health = 75

// Const (cannot be reassigned)
const MAX = 100

Types

  • Integer — whole numbers: 42
  • Float — decimals: 3.14
  • String — text: "hello"
  • Boolean — true or false
  • Null — null
  • Array — [1, 2, 3]
  • Map — { name: "Hero" }
  • Range — 1..10
  • Function — function greet(name) { }

Operators

Arithmetic

+  -  *  /  %

Comparison

==  !=  <  >  <=  >=

Logical

and  or  not

Range

1..10  // exclusive range

Strings

name = "ALX"

// Concatenation
greeting = "Hello, " + name + "!"

// Interpolation
greeting = "Hello, {name}!"

// String length
len = greeting.length

Arrays

Ordered collections of values.

// Create
numbers = [1, 2, 3, 4, 5]
fruits = ["apple", "banana", "cherry"]

// Index (0-based)
first = numbers[0]     // 1
third = fruits[2]      // "cherry"

// Length
len = numbers.length    // 5

// Mutating methods
numbers.push(6)        // add to end
last = numbers.pop()    // remove & return last
numbers.reverse()       // reverse in place

// Access methods
first = numbers.first   // first element
last = numbers.last     // last element

// Search
has3 = numbers.contains(3)    // true
idx = numbers.indexOf(3)     // index or -1

// Join
str = numbers.join("-")       // "1-2-3"

// Index assignment
numbers[0] = 99

// Mixed types
mixed = [1, "hello", true, null]

// Nested
matrix = [[1, 2], [3, 4]]
val = matrix[1][0]   // 3

Maps

Key-value collections (dictionaries).

// Create with identifier keys
player = {
    name: "Hero",
    health: 100,
    speed: 5.5
}

// Create with string keys
settings = {
    "volume": 80,
    "fullscreen": true
}

// Dot access
print(player.name)       // "Hero"

// Index access
print(player["name"])    // "Hero"

// Dot assignment
player.health = 75

// Methods
len = player.length          // 3
k = player.keys()            // ["name", "health", "speed"]
v = player.values()          // ["Hero", 100, 5.5]
has = player.containsKey("name")  // true
val = player.get("mana", 0)       // 0 (default)

// Nested
enemies = [
    { name: "Zombie", health: 50 },
    { name: "Skeleton", health: 30 }
]
print(enemies[0].name)  // "Zombie"

Conditions

health = 100

if health <= 0 {
    print("Dead")
} else if health > 50 {
    print("Healthy")
} else {
    print("Weak")
}

Loops

While

x = 5
while x > 0 {
    print(x)
    x = x - 1
}

For (ranges)

for i in 1..10 {
    print(i)  // 1, 2, ... 9
}

// Break and continue
for i in 1..100 {
    if i == 5 { continue }
    if i == 10 { break }
    print(i)
}

Functions

ALX supports both function and the shorter afun keyword for declaring functions. Both are equivalent.

// Declaration using afun (preferred)
afun greet(name) {
    print("Hello, {name}!")
}

// Declaration using function (still works)
function legacyGreet(name) {
    print("Hello, {name}!")
}

// Call
greet("World")
legacyGreet("World")

// Return values
afun add(a, b) {
    return a + b
}

result = add(10, 20)

// Expression-bodied functions (simple case)
afun addOne(x) = x + 1

// Recursion
afun factorial(n) {
    if n <= 1 { return 1 }
    return n * factorial(n - 1)
}

Lambda Expressions

// Lambda syntax
square = lambda(x) { x * x }
print(square(7))   // 49

// Higher-order functions
afun apply(func, value) {
    return func(value)
}

print(apply(square, 5))  // 25

// Closures
afun makeCounter() {
    count = 0
    afun increment() {
        count = count + 1
        return count
    }
    return increment
}

counter = makeCounter()
print(counter())  // 1
print(counter())  // 2

User Input

Read user input from the terminal with input().

// With a prompt
name = input("What is your name? ")
print("Hello, {name}!")

// Without a prompt
line = input()

// input() always returns a string
age = input("Age: ")
print("You are {age} years old")

Error Handling
