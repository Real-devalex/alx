Arrays

Ordered collections of values.

// Create
numbers = [1, 2, 3, 4, 5]

// Index (0-based)

// Length
len = numbers.length    // 5

// Methods
numbers.push(6)
last = numbers.pop()
first = numbers.first
last = numbers.last
has3 = numbers.contains(3)
idx = numbers.indexOf(3)
str = numbers.join("-")
numbers.reverse()
numbers[0] = 99

Maps

Key-value collections (dictionaries).

// Create
player = { name: "Hero", health: 100 }

// Access
print(player.name)       // Hero
print(player["name"])    // Hero

// Assign
player.health = 75

// Methods
player.length            // 1
player.keys()            // ["name", "health"]
player.values()          // ["Hero", 75]
player.containsKey("name")
player.get("mana", 0)

Conditions

health = 100

if health <= 0 {
    print("Dead")
} else if health > 50 {
    print("Healthy")
} else {
    print("Weak")
}

Loops

// While
x = 5
while x > 0 {
    print(x)
    x = x - 1
}

// For (ranges)
for i in 1..10 {
    if i == 5 { continue }
    if i == 8 { break }
    print(i)
}

Functions

// afun (preferred) or function
afun greet(name) {
    print("Hello, {name}!")
}

afun add(a, b) = a + b  // expression-bodied

afun factorial(n) {
    if n <= 1 { return 1 }
    return n * factorial(n - 1)
}

Lambdas & Closures

square = lambda(x) { x * x }
print(square(7))  // 49

// Closures capture outer scope
afun makeCounter() {
    count = 0
    afun increment() {
        count = count + 1
        return count
    }
    return increment
}
counter = makeCounter()
print(counter())  // 1
print(counter())  // 2

Classes & Objects

Create reusable types with classes, constructors, methods, and inheritance.

// Define a class
class Player {
    function constructor(name, health) {
        this.name = name
        this.health = health
    }

    function takeDamage(amount) {
        this.health = this.health - amount
        print("{this.name} took {amount} damage!")
    }

    function isAlive() {
        return this.health > 0
    }
}

// Create instances
hero = new Player("Hero", 100)
hero.takeDamage(30)      // Hero took 30 damage!
print(hero.isAlive())     // true

Inheritance

class Entity {
    function constructor(name, health) {
        this.name = name
        this.health = health
    }

    function describe() {
        return "{this.name} (HP: {this.health})"
    }
}

class Zombie extends Entity {
    function constructor(name, health, damage) {
        super(name, health)
        this.damage = damage
    }

    function attack(target) {
        target.takeDamage(this.damage)
    }
}

z = new Zombie("Bob", 50, 15)
print(z.describe())  // Bob (HP: 50)
