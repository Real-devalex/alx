Functions

// afun is the preferred shorter keyword
afun greet(name) {
    print("Hello, {name}!")
}

greet("Daniel")

// function keyword also works
function add(a, b) {
    return a + b
}

print(add(10, 20))  // 30

// Expression-bodied functions
afun double(x) = x * 2
print(double(5))  // 10

// Recursion
afun factorial(n) {
    if n <= 1 { return 1 }
    return n * factorial(n - 1)
}

print(factorial(5))  // 120

While Loops

health = 100

while health > 0 {
    health = health - 25
    print(health)
}

For Loops

// Basic for loop
for i in 1..5 {
    print(i)
}

// Break — exit early
for i in 1..10 {
    if i == 5 { break }
    print(i)
}

// Continue — skip iterations
for i in 1..6 {
    if i % 2 == 0 { continue }
    print(i)
}

// Sum
total = 0
for i in 1..10 {
    total = total + i
}
print(total)  // 45

Lambdas & Closures

// Lambda expressions
square = lambda(x) { x * x }
print(square(7))   // 49

// Higher-order functions
afun apply(func, value) {
    return func(value)
}
print(apply(square, 5))  // 25

// Closures — functions that capture outer variables
afun makeCounter() {
    count = 0
    afun increment() {
        count = count + 1
        return count
    }
    return increment
}

counter = makeCounter()
print(counter())  // 1
print(counter())  // 2
print(counter())  // 3

Game Example

// RPG Character System
afun createCharacter(name, health) {
    return {
        name: name,
        health: health,
        maxHealth: health
    }
}

afun takeDamage(char, amount) {
    char.health = char.health - amount
    if char.health < 0 {
        char.health = 0
    }
    print("{char.name} took {amount} damage! HP: {char.health}/{char.maxHealth}")
}

afun heal(char, amount) {
    char.health = char.health + amount
    if char.health > char.maxHealth {
        char.health = char.maxHealth
    }
    print("{char.name} healed {amount}! HP: {char.health}/{char.maxHealth}")
}

// Create a warrior
warrior = createCharacter("Warrior", 100)
mage = createCharacter("Mage", 60)

// Battle
takeDamage(warrior, 30)
takeDamage(mage, 20)
heal(warrior, 15)

// Combat loop
turn = 0
while warrior.health > 0 and mage.health > 0 {
    turn = turn + 1
    takeDamage(mage, 25)
    takeDamage(warrior, 15)
    if turn > 10 { break }
}

if warrior.health > 0 {
    print("Warrior wins!")
} else {
    print("Mage wins!")
}
