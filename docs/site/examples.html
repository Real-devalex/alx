Complete, runnable ALX programs to learn from.

Hello World

// Hello World in ALX
print("Hello, ALEXION!")

Variables

name = "ALEXION STUDIOS"
version = 1
health = 100
speed = 5.5
alive = true

print(name)     // ALEXION STUDIOS
print(version)  // 1
print(health)   // 100
print(speed)    // 5.5
print(alive)    // true

Math

a = 10
b = 20
print(a + b)     // 30

// Operator precedence
print(10 + 5 * 2)   // 20 (not 30)

// Variable reassignment
health = 100
damage = 25
health = health - damage
print(health)    // 75

Conditions

health = 100

if health <= 0 {
    print("Player died")
} else if health > 50 {
    print("Healthy")
} else {
    print("Weak")
}

// Boolean logic
is_active = true
has_permission = false

if is_active and has_permission {
    print("Full access")
} else if is_active {
    print("Limited access")
}

Functions

function greet(name) {
    print("Hello, {name}!")
}

greet("Daniel")

// Return values
function add(a, b) {
    return a + b
}

print(add(10, 20))  // 30

// Recursion
function factorial(n) {
    if n <= 1 { return 1 }
    return n * factorial(n - 1)
}

print(factorial(5))  // 120

While Loops

health = 100

while health > 0 {
    health = health - 25
    print(health)
}
// 75, 50, 25, 0

For Loops

// Basic for loop
for i in 1..5 {
    print(i)
}

// Break — exit early
for i in 1..10 {
    if i == 5 { break }
    print(i)
}

// Continue — skip iterations
for i in 1..6 {
    if i % 2 == 0 { continue }
    print(i)
}

// Sum
total = 0
for i in 1..10 {
    total = total + i
}
print(total)  // 45

String Interpolation

name = "ALX"
version = 2

print("Welcome to {name} {version}!")
// Welcome to ALX 2!

x = 10
y = 5
print("{x} + {y} = {x + y}")
// 10 + 5 = 15

Game Example

// Simple combat system
health = 100
damage = 30
heal = 15

// Take damage
health = health - damage
print("After hit: {health}")  // 70

// Heal
health = health + heal
print("After heal: {health}")  // 85

// Combat loop
turns = 0
while health > 0 {
    health = health - 20
    turns = turns + 1
    print("Turn {turns}: health = {health}")
}

print("Player survived {turns} turns")
